// =============================================================
// 2. FUNCIÓN DE FECHA SIN DESFASE (SOLO DÍA/MES/AÑO)
// =============================================================
function formatearFecha(fechaISO) {
  if (!fechaISO) return '-';
  
  // Extrae la parte YYYY-MM-DD directamente evitando conversión de zona horaria
  const fechaSolo = String(fechaISO).split('T')[0]; 
  const partes = fechaSolo.split('-');
  
  if (partes.length !== 3) return '-';
  
  const [year, month, day] = partes;
  return `${day}/${month}/${year}`;
}

// =============================================================
// 3. CARGA Y CONSULTA DE VALES
// =============================================================
async function cargarOrdenes() {
  const tbody = document.getElementById('tablaCuerpo');

  try {
    const { data, error } = await client
      .from('ordenes')
      .select('*')
      .order('id', { ascending: false });

    if (error) throw error;

    ordenesLocales = data || [];
    ordenesFiltradas = [...ordenesLocales];
    renderizarTabla(ordenesFiltradas);

    // Suscribirse en tiempo real solo la primera vez
    if (primerCarga) {
      escucharNuevosVales();
      primerCarga = false;
    }

  } catch (err) {
    tbody.innerHTML = `
      <tr>
        <td colspan="10" class="text-center py-4 text-danger fw-semibold">
          Error al consultar los registros: ${err.message}
        </td>
      </tr>`;
  }
}

// Renderizar Filas de la Tabla
function renderizarTabla(lista, nuevoId = null) {
  const tbody = document.getElementById('tablaCuerpo');

  if (!lista || lista.length === 0) {
    tbody.innerHTML = `
      <tr>
        <td colspan="10" class="text-center py-4 text-muted fs-6">
          No se encontraron Vales registrados.
        </td>
      </tr>`;
    return;
  }

  let html = '';

  lista.forEach(item => {
    const esNuevo = item.id === nuevoId;
    const claseResaltado = esNuevo ? 'nueva-fila' : '';
    const fechaTexto = formatearFecha(item.created_at || item.fecha);

    html += `
      <tr class="${claseResaltado}" id="fila-${item.id}">
        <td class="ps-4 fw-bold text-primary">#${item.id}</td>
        
        <!-- FECHA SIN DESFASE Y LEGIBLE -->
        <td class="fw-bold text-dark text-nowrap" style="font-size: 0.9rem;">${fechaTexto}</td>
        
        <td class="fw-semibold text-dark">${item.nombre_empresa || '-'}</td>
        <td>${item.marca || '-'}</td>
        <td><span class="badge-patente">${item.patente || '-'}</span></td>
        <td class="text-muted text-truncate" style="max-width: 180px;">${item.detalle || '-'}</td>
        <td>
          <span class="badge text-bg-light border text-secondary fw-normal">
            ${item.medio_pago || 'N/A'}
          </span>
        </td>
        <td class="text-end text-muted font-monospace">$${(item.valor_neto || 0).toLocaleString('es-CL')}</td>
        <td class="text-end fw-bold text-dark font-monospace">$${(item.valor_total || 0).toLocaleString('es-CL')}</td>
        <td class="text-center pe-4">
          <div class="btn-group btn-group-sm" role="group">
            <button onclick="abrirModalEditar(${item.id})" class="btn btn-outline-secondary border-0" title="Editar Vale">
              <i class="bi bi-pencil text-primary"></i>
            </button>
            <button onclick="eliminarVale(${item.id})" class="btn btn-outline-secondary border-0" title="Eliminar Vale">
              <i class="bi bi-trash text-danger"></i>
            </button>
          </div>
        </td>
      </tr>
    `;
  });

  tbody.innerHTML = html;
}

// 4. TIEMPO REAL Y ALERTAS (INSERT, UPDATE, DELETE)
// =============================================================
function escucharNuevosVales() {
  client
    .channel('realtime_ordenes')
    .on('postgres_changes', { event: '*', schema: 'public', table: 'ordenes' }, payload => {
      if (payload.eventType === 'INSERT') {
        const nuevoVale = payload.new;
        ordenesLocales.unshift(nuevoVale);
        filtrarTabla(document.getElementById('inputBuscar').value.toLowerCase().trim(), nuevoVale.id);
        mostrarToast(`Folio #${nuevoVale.id} - ${nuevoVale.patente || 'Sin Patente'}`);
      } 
      else if (payload.eventType === 'UPDATE') {
        const index = ordenesLocales.findIndex(item => item.id === payload.new.id);
        if (index !== -1) {
          ordenesLocales[index] = payload.new;
          filtrarTabla(document.getElementById('inputBuscar').value.toLowerCase().trim());
        }
      }
      else if (payload.eventType === 'DELETE') {
        ordenesLocales = ordenesLocales.filter(item => item.id !== payload.old.id);
        filtrarTabla(document.getElementById('inputBuscar').value.toLowerCase().trim());
      }
    })
    .subscribe();
}

function mostrarToast(mensaje) {
  document.getElementById('toastTexto').innerText = mensaje;
  const toastEl = document.getElementById('toastNuevoVale');
  const toast = new bootstrap.Toast(toastEl);
  toast.show();
}

  const vale = ordenesLocales.find(item => item.id === id);
  if (!vale) return;

  document.getElementById('modalEditarLabel').innerText = `Editar Vale #${vale.id}`;
  document.getElementById('editId').value = vale.id;
  document.getElementById('editEmpresa').value = vale.nombre_empresa || '';
  document.getElementById('editMedioPago').value = vale.medio_pago || 'Efectivo';
  document.getElementById('editMarca').value = vale.marca || '';
  document.getElementById('editPatente').value = vale.patente || '';
  document.getElementById('editDetalle').value = vale.detalle || '';
  document.getElementById('editValorTotal').value = vale.valor_total || 0;
  document.getElementById('editValorNeto').value = vale.valor_neto || 0;

  if (!modalEditarBs) {
    modalEditarBs = new bootstrap.Modal(document.getElementById('modalEditar'));
  }
  modalEditarBs.show();
}

// Guardar Edición
formEditar.addEventListener('submit', async (e) => {
  e.preventDefault();

  const id = parseInt(document.getElementById('editId').value);
  const total = parseFloat(inputTotal.value) || 0;
  const neto = parseFloat(inputNeto.value) || 0;

  const datosActualizados = {
    nombre_empresa: document.getElementById('editEmpresa').value,
    medio_pago: document.getElementById('editMedioPago').value,
    marca: document.getElementById('editMarca').value.toUpperCase().trim(),
    patente: document.getElementById('editPatente').value.toUpperCase().trim(),
    detalle: document.getElementById('editDetalle').value.trim(),
    valor_total: total,
    valor_neto: neto
  };

  try {
    const { error } = await client
      .from('ordenes')
      .update(datosActualizados)
      .eq('id', id);

    if (error) throw error;

    modalEditarBs.hide();
    
    Swal.fire({
      icon: 'success',
      title: '¡Vale Actualizado!',
      text: `El folio #${id} ha sido actualizado correctamente.`,
      timer: 2000,
      showConfirmButton: false
    });

  } catch (err) {
    Swal.fire({
      icon: 'error',
      title: 'Error',
      text: 'No se pudo actualizar el vale: ' + err.message
    });
  }
});

// Eliminar Vale
async function eliminarVale(id) {
  const confirmacion = await Swal.fire({
    title: `¿Eliminar Vale #${id}?`,
    text: "Esta acción no se puede deshacer.",
    icon: 'warning',
    showCancelButton: true,
    confirmButtonColor: '#dc3545',
    cancelButtonColor: '#6c757d',
    confirmButtonText: 'Sí, eliminar',
    cancelButtonText: 'Cancelar'
  });

  if (!confirmacion.isConfirmed) return;

  try {
    const { data: { user } } = await client.auth.getUser();

    if (!user) {
      Swal.fire({
        icon: 'error',
        title: 'Acceso Denegado',
        text: 'Debes iniciar sesión con una cuenta de administrador autorizada para eliminar vales.'
      });
      return;
    }

    const { error } = await client
      .from('ordenes')
      .delete()
      .eq('id', id);

    if (error) throw error;

    await client.rpc('reset_secuencia_ordenes');

    Swal.fire({
      icon: 'success',
      title: 'Eliminado',
      text: `El vale #${id} ha sido borrado correctamente.`,
      timer: 1800,
      showConfirmButton: false
    });

  } catch (err) {
    Swal.fire({
      icon: 'error',
      title: 'Error al eliminar',
      text: err.message
    });
  }
}

// =============================================================
// 6. BUSCADOR Y EXPORTACIÓN A EXCEL
// =============================================================
function filtrarTabla(termino, nuevoId = null) {
  ordenesFiltradas = ordenesLocales.filter(item => {
    return (
      (item.patente && item.patente.toLowerCase().includes(termino)) ||
      (item.nombre_empresa && item.nombre_empresa.toLowerCase().includes(termino)) ||
      (item.marca && item.marca.toLowerCase().includes(termino)) ||
      (item.id && item.id.toString().includes(termino))
    );
  });
  renderizarTabla(ordenesFiltradas, nuevoId);
}

  <script>
    const SUPABASE_URL = 'https://okkydrwdexacxvfbtopb.supabase.co'; 
    const SUPABASE_ANON_KEY = 'sb_publishable_XY2XRiAQw4tQdygSAwd4XA_pRETQ_EA';              
    const client = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    let valesLocales = [];

    // 1. Control de Autenticación al Cargar
    document.addEventListener('DOMContentLoaded', async () => {
      const { data: { session } } = await client.auth.getSession();

      if (session) {
        mostrarAdmin();
      } else {
        mostrarLogin();
      }
    });

    // Escuchar cambios de estado en la autenticación
    client.auth.onAuthStateChange((event, session) => {
      if (session) {
        mostrarAdmin();
      } else {
        mostrarLogin();
      }
    });

    function mostrarAdmin() {
      document.getElementById('loginSection').classList.add('hidden');
      document.getElementById('adminSection').classList.remove('hidden');
      cargarVales();
    }

    function mostrarLogin() {
      document.getElementById('adminSection').classList.add('hidden');
      document.getElementById('loginSection').classList.remove('hidden');
    }

    // 2. Evento del Login Form
    document.getElementById('formLogin').addEventListener('submit', async (e) => {
      e.preventDefault();
      const email = document.getElementById('loginEmail').value.trim();
      const password = document.getElementById('loginPassword').value.trim();
      const btnLogin = document.getElementById('btnLogin');

      btnLogin.disabled = true;
      btnLogin.innerText = 'VALIDANDO...';

      try {
        const { data, error } = await client.auth.signInWithPassword({ email, password });
        if (error) throw error;
        
        mostrarAdmin();
      } catch (err) {
        Swal.fire({
          icon: 'error',
          title: 'Error de Acceso',
          text: err.message === 'Invalid login credentials' ? 'Correo o contraseña incorrectos.' : err.message,
          customClass: { popup: 'swal2-popup-accessible' }
        });
      } finally {
        btnLogin.disabled = false;
        btnLogin.innerText = 'INICIAR SESIÓN';
      }
    });

    // 3. Cargar Vales desde Supabase
    async function cargarVales() {
      const tbody = document.getElementById('tablaValesBody');
      tbody.innerHTML = `
        <tr id="filaCargando">
          <td colspan="9" class="py-12 text-center text-slate-500 font-bold">
            <div class="inline-block animate-spin rounded-full h-8 w-8 border-4 border-blue-600 border-t-transparent mb-3"></div>
            <p>Cargando Vales de Servicio...</p>
          </td>
        </tr>
      `;

      try {
        const { data: vales, error } = await client
          .from('ordenes')
          .select('*')
          .order('id', { ascending: false });

        if (error) throw error;

        valesLocales = vales || [];
        renderizarTabla(valesLocales);

      } catch (err) {
        console.error('Error:', err);
        tbody.innerHTML = `
          <tr>
            <td colspan="9" class="py-8 text-center text-red-600 font-bold">
              ❌ Error al obtener los vales: ${err.message}
            </td>
          </tr>
        `;
      }
    }

    // 4. Renderizar Filas de la Tabla
    function renderizarTabla(lista) {
      const tbody = document.getElementById('tablaValesBody');

      if (!lista || lista.length === 0) {
        tbody.innerHTML = `
          <tr>
            <td colspan="9" class="py-8 text-center text-slate-400 font-bold">
              No se encontraron vales de servicio registrados.
            </td>
          </tr>
        `;
        return;
      }

      tbody.innerHTML = lista.map(vale => {
        const fecha = vale.created_at ? new Date(vale.created_at).toLocaleDateString('es-CL', {
          day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit'
        }) : 'N/A';

        const netoFormatted = (vale.valor_neto || 0).toLocaleString('es-CL');
        const totalFormatted = (vale.valor_total || 0).toLocaleString('es-CL');

        return `
          <tr class="hover:bg-slate-50 transition-all border-b border-slate-200">
            <td class="py-3.5 px-4 text-center font-black text-blue-600">#${vale.id}</td>
            <td class="py-3.5 px-4 text-xs font-semibold text-slate-500">${fecha}</td>
            <td class="py-3.5 px-4 font-bold text-slate-900">${vale.nombre_empresa || '-'}</td>
            <td class="py-3.5 px-4 uppercase font-bold text-slate-800">${vale.marca || '-'}</td>
            <td class="py-3.5 px-4 font-black tracking-wider text-blue-900 uppercase">${vale.patente || '-'}</td>
            <td class="py-3.5 px-4 text-xs text-slate-600 max-w-xs truncate">${vale.detalle || '-'}</td>
            <td class="py-3.5 px-4 text-xs font-bold text-slate-700">${vale.medio_pago || '-'}</td>
            <td class="py-3.5 px-4 text-right font-bold text-slate-700">$${netoFormatted}</td>
            <td class="py-3.5 px-4 text-right font-black text-slate-900">$${totalFormatted}</td>
          </tr>
        `;
      }).join('');
    }

    // 5. Filtro de Búsqueda Rápida
    function filtrarTabla() {
      const query = document.getElementById('inputBuscar').value.toLowerCase().trim();
      const filtrados = valesLocales.filter(vale => {
        return (vale.id && vale.id.toString().includes(query)) ||
               (vale.patente && vale.patente.toLowerCase().includes(query)) ||
               (vale.nombre_empresa && vale.nombre_empresa.toLowerCase().includes(query)) ||
               (vale.marca && vale.marca.toLowerCase().includes(query));
      });
      renderizarTabla(filtrados);
    }

    // 6. Cerrar Sesión
    async function cerrarSesionAdmin() {
      await client.auth.signOut();
      mostrarLogin();
    }
